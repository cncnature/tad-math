# 단계별 문제 스키마 — `data/problems/<파일>.js` (렌더러 js/app.js `sub()`), 단원 노트 — `data/unit-notes/<파일>.js`

왜: 확인 퀴즈(기본)만으로는 내신 상위권·모의고사 1등급에 못 간다. 소단원마다 **① 기본(확인 퀴즈) → ② 실력(내신 중간·기말형) → ③ 1등급 도전(학평·수능형 자료 해석·서술형)** 의 사다리를 둔다(바람직한 어려움 · 인출 연습 · 자료 해석 전이).

```js
window.DATA = window.DATA || {}; (window.DATA.problemsAdd = window.DATA.problemsAdd || []).push({
 "<차시 id>": [
  { lv:2, kind:"mc", tag:"내신 중간·기말형", q:"문제 (자작, ≤300자, **굵게**·$수식$ 가능)", opts:["①","②","③","④"], a:0, exp:"해설 (왜 정답이고 왜 오답인지)", err:"개념" },
  { lv:3, kind:"mc", tag:"학평·수능형", data:"<자료> 표·그래프 설명·사료·지문 (markdown, 표는 | 구분, ≤400자)", q:"자료에 대한 설명으로 옳은 것은?", opts:["…","…","…","…"], a:2, exp:"…", why3:"1등급 포인트 — 보기 소거 기준 한 줄", err:"자료" },
  { lv:3, kind:"essay", tag:"서술형", q:"…을 두 가지 근거를 들어 서술하시오.", model:"모범 답안 요지(2~4문장)", rubric:[["근거 ① 제시",3],["근거 ② 제시",3],["용어 정확",2],["문장 완결",2]], exp:"채점 포인트" }
 ]
});
```
- 차시마다 **lv2 2문항 + lv3 2문항(객관식 1 + 서술형 1)** 이 기본. `err` 는 오답 유형 태그: `개념` `자료` `계산` `실수` `어휘` `추론` 중 하나.
- `kind:"mc"` 는 4지선다·정답 하나, `kind:"essay"` 는 `model`·`rubric`(합 10점) 필수. 교과서·문제집·기출 원문을 베끼지 않는다. 국어·영어 지문은 짧은 자작 지문을 `data` 에 넣는다.
- 같은 차시에 여러 파일이 써도 합쳐진다(js/merge.js). 검사: `node scripts/check.js`.

단원 노트(학습 목표 · 교과서 핵심 요약 · 시험 경향):
```js
window.DATA = window.DATA || {}; (window.DATA.unitNotesAdd = window.DATA.unitNotesAdd || []).push({
 "<단원 id>": { goals:["학습 목표 3~4개 — 성취기준을 학생 말로('~을 설명할 수 있다')"], summary:["교과서 핵심 요약 6~10줄, 각 ≤90자, **굵게**"], keywords:["핵심 용어 8~12개"], examTips:["중간·기말 출제 경향 2~3줄","학평·수능 연결 1~2줄"] }
});
```
