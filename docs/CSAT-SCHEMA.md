# 수능 전략 센터 데이터 형식 (data/csat/*.js)

`#/csat` 화면(js/csat.js)이 읽는 데이터다. 대상은 2026년 중2 학생이고, 이 학생은 **2031학년도 수능(2030년 11월)**을 치른다. 기준은 **2028학년도 이후 통합형 수능**이다. 2023-12-27 교육부 확정안과 2025-01-20 평가원 문항 수·점수 체제 발표를 따른다.

- **국어:** 화법과 언어, 독서와 작문, 문학 (45문항, 80분)
- **수학:** 대수, 미적분Ⅰ, 확률과 통계 (30문항 중 단답 9, 100분)
- **영어:** 영어Ⅰ·Ⅱ (45문항 중 듣기 17, 70분, 절대평가)
- **한국사:** 20문항, 30분, 절대평가
- **통합사회·통합과학:** 과목별 25문항, 40분, 배점 1.5·2·2.5점
- **성적:** 9등급 상대평가 유지 (영어·한국사·제2외국어/한문은 절대평가)

## 1. data/csat/system.js — 체계·일정·정책·경향

```js
window.DATA = window.DATA || {};
window.DATA.csat = {
  updated: "YYYY-MM-DD",
  target: { examYear: 2031, examDate: "2030-11-14", note: "추정 날짜" },
  areas: [ { id, name, icon, col, items, minutes, scope:[...], grading, note } ],
  stages: [ { id:"m2", name, from, to, focus, tasks:{ <areaId>: [ {t, h} ] } } ],  // 학년별 학습 프로그램
  policy: [ { date, title, status:"확정"|"논의"|"예정", impact, src } ],         // 제도 변경 감시
  trends: [ { year:"2027 수능", date, summary, byArea:{ <areaId>: "..." }, hard:[ {area, no, type} ], src:[...] } ]
};
```

## 2. data/csat/<영역>.js — 출제 패턴

각 파일은 먼저 자기 영역의 출제 갈래 트리를 등록하고, 이어서 패턴을 배열에 추가한다. 파일마다 독립적으로 동작해야 한다. 갈래는 2022 개정 과목의 내용 체계를 따르고, 영역마다 4~9개로 나눈다.

```js
window.DATA = window.DATA || {};
(window.DATA.csatTree = window.DATA.csatTree || {}).kor = [
  { id:"k-read-soc", name:"독서 · 사회·법·경제", desc:"60자 이내 설명", subject:"독서와 작문" }
];
(window.DATA.csatPatterns = window.DATA.csatPatterns || []).push(
 { id:"k-dok-law-apply",   // 고유 id: 영역 접두어(k·m·e·h·s·c) + 갈래 + 짧은 이름
   area:"kor",             // kor | math | eng | his | soc | sci
   strand:"k-read-soc",    // 위 csatTree[area] 의 id
   name:"법·제도 지문의 사례 적용",   // 20자 이내
   slot:"독서 세트 마지막 문항 · 3점",    // 문항 번호대·배점(2028 체계 기준, 추정이면 '추정')
   weight:3,               // 중요도 1~3 (3 = 매년 나오고 변별력이 큼)
   status:"계속",          // 계속 | 강화 | 신유형(2028 예시) | 축소
   years:["2025","2026","2027","예시"],  // 대표 출제 학년도(예시 = 2028 예시문항). 확인된 것만
   start:"m3",             // 준비를 시작할 학년 단계: m2 | m3 | h1 | h2 | h3
   signal:["...", "..."],  // 문제에서 이 패턴을 알아보는 신호 2~3개, 각 60자 이내
   first:"...",            // 첫 수(가장 먼저 할 행동) 120자 이내. **굵게**, $수식$ 가능
   steps:["...", "..."],   // 풀이 절차 2~4단계, 각 120자 이내
   trap:"...",             // 대표 함정 100자 이내
   prep:{ m2:"...", m3:"...", h1:"...", h2:"...", h3:"..." },  // 학년별 준비 방법(해당 없는 학년은 생략), 각 80자 이내
   links:[ { t:"사이트 차시 제목", h:"#/sub/kor/kor-h2-dokjak-set" } ],  // 이 사이트 안의 실제 페이지만(파일에서 id 를 확인)
   q:"...",                // 자작 동형 문제(원문 전재 금지). 자료가 필요하면 q 안에 \n 으로 포함
   opts:["①","②","③","④","⑤"],  // 5지선다(수학 단답형이면 opts 생략하고 ans 에 정답)
   a:0,                    // 정답 인덱스 0~4 (파일 전체에서 고르게 분산)
   ans:"",                 // 단답형 정답(수학만)
   exp:"...",              // 해설 200자 이내
   src:[ ["제목","URL"] ]  // 실제로 연 출처(기사·평가원 발표·분석 자료). 최소 1개
 }
);
```

## 3. 규칙

- **저작권:** 기출·EBS·사교육 문항과 지문 원문은 쓰지 않는다. 문제는 모두 자작하고, 소재·구조만 참고한다.
- **2028 범위 밖 내용:** 미적분Ⅱ, 기하, 언어와 매체 단독 과목처럼 2028 범위에서 빠진 내용은 패턴으로 만들지 않는다. 다만 status 를 "축소"로 두고 이유를 적는 것은 가능하다.
- **신유형:** 2028 예시문항에서 새로 나온 유형은 status 를 "신유형"으로 둔다. 근거 출처는 반드시 적는다.
- **출제 연도·문항 번호:** 연도별 출제 여부나 문항 번호를 모르면 years 에 넣지 않는다. 지어내지 않는다.
- **글자 수:** 위에 적은 한도를 지킨다. 학생이 화면에서 한눈에 볼 수 있어야 한다.
